document.getElementById('login-form').addEventListener('submit', async function(event) {
        event.preventDefault();
        const username = document.getElementById('username').value;
        const password = document.getElementById('password').value;
        
        try {
            const response = await fetch('http://127.0.0.1:8000/login/token', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({
                    username: username,
                    password: password
                })
            });

        if (response.ok) {
            const data = await response.json();
            document.getElementById('message').innerHTML = "¡Inicio de sesión exitoso!";
        } else {
            const errorData = await response.json();
            document.getElementById('message').innerHTML = `Error: ${errorData.message}`;
        }
    } catch (error) {
        document.getElementById('message').innerHTML = 'Hubo un error al intentar iniciar sesión.';
        console.error('Error:', error);
    }
    });
